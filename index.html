COMPROG TRANSPORTATION — Fare calculator with student discount, validation, and receipt.

#include <iostream>
#include <iomanip>
#include <string>
#include <cctype>
#include <cstdlib>
using namespace std;

bool isNumber(string s) {
    if(s.empty()) return false;
    for(int i=0; i<s.length(); i++) {
        if(!isdigit(s[i])) return false;
    }
    return true;
}

int main() {
    string destName[] = {"", "COGEO", "PADILLA", "BLUE MOUNTAIN", "CORAZON", "PANDAYAN", "RC", "PAGRAI", "FRANCIA"};

        cout << " 4 - CORAZON \t\t 8 - FRANCIA" << endl;
        cout << " 9 - EXIT" << endl;

        while(true) {
            cout << "\nENTER YOUR DESTINATION : ";
            string d; cin >> d;
            if(!isNumber(d)) { cout << "INVALID! 1-9 ONLY!\n"; continue; }
            dest = atoi(d.c_str());
            if(dest == 9) { cout << "PROGRAM CLOSED. THANK YOU!"; return 0; }
            if(dest < 1 || dest > 8) { cout << "INVALID! 1-8 ONLY!\n"; continue; }
            break;
        }

        cout << "STUDENT (Y/N)? : "; cin >> stud;
        double fare = *(fareList + dest);
        double discount = (stud=='Y' || stud=='y')? fare * 0.20 : 0.00;
        double total = fare - discount;

        double amount = 0; string amountStr;
        while(true) {
            cout << "ENTER AMOUNT PAID : "; cin >> amountStr;
            if(!isNumber(amountStr)) { cout << "NUMBERS ONLY!\n"; continue; }
            amount = atof(amountStr.c_str());
            if(amount < total) { cout << "KULANG! Total is " << total << endl; continue; }
            break;
        }
